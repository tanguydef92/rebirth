cmake_minimum_required(VERSION 3.16)
project(gazebo_swarm_beta LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(SwarmPlayback SHARED src/SwarmPlayback.cc)
target_include_directories(SwarmPlayback PRIVATE src)
target_link_libraries(SwarmPlayback PRIVATE gz-sim8::gz-sim8 gz-plugin2::gz-plugin2)
add_executable(trajectory_test tests/trajectory_test.cc)
target_include_directories(trajectory_test PRIVATE src)
target_compile_options(trajectory_test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME trajectory COMMAND trajectory_test)
